CREATE TABLE user (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255),
    auth_key VARCHAR(255)
);

CREATE TABLE location (
    location_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255),
    address VARCHAR(255)
);

CREATE TABLE store_section (
    section_id INT AUTO_INCREMENT PRIMARY KEY,
    location_id INT,
    name VARCHAR(255),
    sort_order INT,
    CONSTRAINT fk_store_section_location FOREIGN KEY (location_id) REFERENCES location(location_id) ON DELETE CASCADE
);

CREATE TABLE item (
    item_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255),
    image_path VARCHAR(255),
    section_id INT,
    location_id INT,
    CONSTRAINT fk_item_section FOREIGN KEY (section_id) REFERENCES store_section(section_id) ON DELETE SET NULL,
    CONSTRAINT fk_item_location FOREIGN KEY (location_id) REFERENCES location(location_id) ON DELETE CASCADE
);

CREATE TABLE shopping_list (
    shopping_list_id INT AUTO_INCREMENT PRIMARY KEY,
    location_id INT,
    paid BOOLEAN DEFAULT FALSE,
    completed_at TIMESTAMP,
    CONSTRAINT fk_shopping_list_location FOREIGN KEY (location_id) REFERENCES location(location_id) ON DELETE CASCADE
);

CREATE TABLE item_status (
    status_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) UNIQUE
);

CREATE TABLE item_in_list (
    item_in_list_id INT AUTO_INCREMENT PRIMARY KEY,
    item_id INT,
    shopping_list_id INT,
    buyer_id INT,
    quantity INT,
    discount_price INT,
    status_id INT,
    CONSTRAINT fk_item_in_list_item FOREIGN KEY (item_id) REFERENCES item(item_id) ON DELETE CASCADE,
    CONSTRAINT fk_item_in_list_list FOREIGN KEY (shopping_list_id) REFERENCES shopping_list(shopping_list_id) ON DELETE CASCADE,
    CONSTRAINT fk_item_in_list_buyer FOREIGN KEY (buyer_id) REFERENCES user(user_id) ON DELETE SET NULL,
    CONSTRAINT fk_item_in_list_status FOREIGN KEY (status_id) REFERENCES item_status(status_id)
);

CREATE TABLE item_price_history (
    item_price_history_id INT AUTO_INCREMENT PRIMARY KEY,
    item_id INT,
    price INT,
    effective_date TIMESTAMP,
    CONSTRAINT fk_item_price_history_item FOREIGN KEY (item_id) REFERENCES item(item_id) ON DELETE CASCADE
);
